```python
import time


class TokenBucket:
    """Lazy-refill token bucket rate limiter."""

    def __init__(self, rate: float, capacity: int) -> None:
        self.rate, self.capacity = rate, capacity
        self.credits, self.stamp = float(capacity), 0.0

    def allow(self, tokens: int = 1) -> bool:
        now = time.monotonic()
        self.credits = min(self.capacity, self.credits + (now - self.stamp) * self.rate)
        self.stamp = now
        if self.credits < tokens:
            return False
        self.credits -= tokens
        return True
```
